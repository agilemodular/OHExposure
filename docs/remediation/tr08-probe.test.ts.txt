// TR-08 probe on the reviewed code (b5eb070): records outcomes only; asserts nothing. Removed after the fix; the output
// is kept in docs/remediation/tr08.before-b5eb070.json.
import { writeFileSync } from 'node:fs';
import { afterAll, it } from 'vitest';
import { closePool } from '../src/db.js';
import * as T from './tr08-scenarios.js';
afterAll(async () => { await closePool(); });
it('TR-08 probe', async () => {
  const S = await T.setup();
  const out: any = { commit: process.env.PROBE_COMMIT, run_at: new Date().toISOString(), scenarios: [] };
  for (const f of [T.heldSample, T.suspensionReview, T.reviewerQcDisposition, T.concurrentAmendment, T.concurrentCalibration]) {
    try { out.scenarios.push(await f(S)); } catch (e: any) { out.scenarios.push({ scenario: f.name, error: e.message }); }
  }
  writeFileSync(process.env.PROBE_OUT!, JSON.stringify(out, null, 2));
}, 120000);
